function classifyWhatsAppMessage(text) {
  const normalized = text.toLowerCase();
  if (/precio|cotiz|costo|presupuesto/.test(normalized)) return "cotizacion";
  if (/falla|problema|soporte|no funciona/.test(normalized)) return "soporte";
  if (/pago|factura|deposito|adeudo/.test(normalized)) return "pago";
  if (/cita|agenda|horario|reservar/.test(normalized)) return "agenda";
  return "general";
}
